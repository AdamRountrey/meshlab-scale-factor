cmake_minimum_required(VERSION 3.18)
project(MeshLabScaleFactor LANGUAGES C CXX)

option(BUILD_MESHLAB_PLUGIN "Build the MeshLab 2025.07 plugin and integration probe" ON)
include(CTest)

add_executable(scale_estimator_tests tests/tests.cpp core/scale_estimator.cpp)
target_include_directories(scale_estimator_tests PRIVATE core)
target_compile_features(scale_estimator_tests PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(scale_estimator_tests PRIVATE /W4)
else()
    target_compile_options(scale_estimator_tests PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME scale_estimator_tests COMMAND scale_estimator_tests)

if(BUILD_MESHLAB_PLUGIN)
    set(MESHLAB_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/meshlab" CACHE PATH "MeshLab 2025.07 source checkout, including vcglib")
    if(NOT EXISTS "${MESHLAB_SOURCE_DIR}/src/vcglib/vcg/complex/complex.h")
        message(FATAL_ERROR "Clone MeshLab-2025.07 and its vcglib submodule into vendor/meshlab, or set MESHLAB_SOURCE_DIR.")
    endif()
    set(MESHLAB_BUILD_MINI ON CACHE BOOL "" FORCE)
    set(MESHLAB_BUILD_ONLY_LIBRARIES ON CACHE BOOL "" FORCE)
    set(MESHLAB_ALLOW_OPTIONAL_EXTERNAL_LIBRARIES OFF CACHE BOOL "" FORCE)
    set(MESHLAB_PLUGINS "")
    add_subdirectory("${MESHLAB_SOURCE_DIR}" meshlab EXCLUDE_FROM_ALL)
    find_package(Qt5 5.15 COMPONENTS Widgets OpenGL Xml Network REQUIRED)
    set(CMAKE_AUTOMOC ON)
    set(CMAKE_AUTOUIC ON)
    set(CMAKE_AUTORCC ON)
    set(MESHLAB_PLUGIN_OUTPUT_DIR "${CMAKE_BINARY_DIR}/plugins")
    set(MESHLAB_PLUGIN_INSTALL_DIR "plugins")
    if(WIN32)
        add_compile_definitions(NOMINMAX _CRT_SECURE_NO_DEPRECATE)
    endif()
    if(MSVC)
        add_compile_options(/bigobj)
    endif()
    add_subdirectory(plugin)
    add_executable(plugin_probe integration/plugin_probe.cpp)
    target_link_libraries(plugin_probe PRIVATE meshlab-common)
    target_compile_features(plugin_probe PRIVATE cxx_std_17)
endif()
